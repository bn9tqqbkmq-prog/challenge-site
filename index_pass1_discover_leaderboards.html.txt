async function initData(){
  localMode = true;
  setModeLabel();

  if(!isFirebaseConfigured()){
    console.log("LOCAL MODE: paste Firebase config to go global.");
    renderAllFromLocal();
    return;
  }

  const CONNECT_TIMEOUT_MS = 2200;
  try{ firebase.initializeApp(firebaseConfig); } catch(e){ /* already init */ }

  const db = firebase.database();
  const room = getRoom();

refs = {
  arc: db.ref(`${room}/arc`),
  totals: db.ref(`${room}/totals`),
  feed: db.ref(`${room}/feed`),
  coins: db.ref(`${room}/coins`),
  decision: db.ref(`${room}/decision`),
  bounties: db.ref(`${room}/bounties`)
};

  const connectPromise = new Promise((resolve, reject) => {
    refs.totals.once("value").then(()=> resolve(true)).catch(reject);
  });
  const timeoutPromise = new Promise(resolve => setTimeout(()=> resolve(false), CONNECT_TIMEOUT_MS));

  let ok = false;
  try{ ok = await Promise.race([connectPromise, timeoutPromise]); } catch(e){ ok = false; }

  if(!ok){
    console.log("Firebase blocked/slow -> staying LOCAL.");
    localMode = true;
    setModeLabel();
    renderAllFromLocal();
    return;
  }

  localMode = false;
  setModeLabel();

  // Seed defaults
  refs.arc.get().then(s => { if(!s.exists()) refs.arc.set(local.arc); });
  refs.totals.get().then(s => { if(!s.exists()) refs.totals.set(local.totals); });
  refs.decision.get().then(s => { if(!s.exists()) refs.decision.set(local.decision); });

  // Listeners
  refs.arc.on("value", snap => {
    const a = snap.val() || local.arc;
    document.getElementById("arcTitle").innerText = a.title || local.arc.title;
    document.getElementById("arcSubtitle").innerText = a.subtitle || local.arc.subtitle;
  });

  refs.totals.on("value", snap => {
    live.totals = snap.val() || local.totals;
    renderTotals(live.totals);
  });

  refs.feed.limitToLast(10).on("value", snap => {
    const obj = snap.val() || {};
    const items = Object.values(obj).sort((a,b)=> (b.ts||0)-(a.ts||0));
    renderFeed(items);
  });

  refs.coins.on("value", snap => {
    live.coinsObj = snap.val() || {};
    renderCoins();
  });

  refs.decision.on("value", snap => {
    live.decision = snap.val() || local.decision;
    renderDecision(live.decision);
    tickCountdown();
    if(live.decision.closed && !window.__announcedWinner){
      window.__announcedWinner = true;
      announceWinner(live.decision);
    }
    if(!live.decision.closed) window.__announcedWinner = false;
  });

  refs.bounties.limitToLast(40).on("value", snap => {
    const obj = snap.val() || {};
    const items = Object.entries(obj).map(([id,v]) => ({ id, ...v }))
      .sort((a,b)=> (b.ts||0)-(a.ts||0));
    live.bounties = items;
    renderBounties(items);
  });
}

function renderDecision(dec){
  document.getElementById("pollQuestion").innerText = dec.question || "Pick the next twist:";
  document.getElementById("pollDesc").innerText = dec.desc || "";

  const container = document.getElementById("pollOptions");
  container.innerHTML = "";

  const options = dec.options || {};
  const totalVotes = Object.values(options).reduce((s,o)=> s + (o.votes||0), 0);

  // find leader
  let leaderKey=null, leaderVotes=-1;
  Object.entries(options).forEach(([k,o])=>{
    const v=o.votes||0;
    if(v > leaderVotes){ leaderVotes=v; leaderKey=k; }
  });

  const leaderEl = document.getElementById("pollLeader");
  if(leaderKey && leaderVotes > 0){
    leaderEl.style.display = "block";
    leaderEl.innerHTML = `<span class="leading">🔥 Leading: ${options[leaderKey].name} (${leaderVotes})</span>`;
  } else {
    leaderEl.style.display = "none";
    leaderEl.innerHTML = "";
  }

  Object.entries(options).forEach(([key,opt])=>{
    const v = opt.votes || 0;
    const pct = totalVotes > 0 ? Math.round((v/totalVotes)*100) : 0;

    const wrap = document.createElement("div");
    wrap.className="pollOpt";

    const head = document.createElement("div");
    head.className="pollHead";

    const left = document.createElement("div");
    left.style.minWidth = "0";
    left.innerHTML = `
      <p class="pollName">${opt.name || key}</p>
      <p class="pollMeta">
        ${dec.closed ? "Closed • " : ""}Votes: ${v} • ${pct}%
        ${key === leaderKey && v > 0 ? " • 🔥 leading" : ""}
      </p>
    `;

    const btn = document.createElement("button");
    btn.className="secondary tabBtn";
    btn.innerText = dec.closed ? "Closed" : `Vote (${VOTE_COST})`;
    btn.disabled = !!dec.closed;
    btn.onclick = () => voteOption(key, VOTE_COST);

    head.appendChild(left);
    head.appendChild(btn);

    const barWrap = document.createElement("div");
    barWrap.className="pollBarWrap";

    const bar = document.createElement("div");
    bar.className="pollBar";
    bar.style.width = pct + "%";

    barWrap.appendChild(bar);

    wrap.appendChild(head);
    wrap.appendChild(barWrap);
    container.appendChild(wrap);
  });
}

function supportAction(type, pounds){
  const name = getUserName();
  if(name === "Anonymous") return alert("Set a display name first.");

  const influence = pounds * INFLUENCE_PER_POUND;
  const msg = type === "time"
    ? `⚡ ${name} supported £${pounds} → +${pounds} minute (+${influence} Influence)`
    : `😈 ${name} supported £${pounds} → comfort removed (+${influence} Influence)`;

  // ✅ LOCAL mode updates
  if(localMode){
    local.totals.total += pounds;
    local.totals.supports += 1;
    if(type === "time") local.totals.time += pounds;
    if(type === "comfort") local.totals.comfort += 1;

    renderTotals(local.totals);
    addMyCoins(influence);
    logAction(msg);
    return;
  }

  // ✅ GLOBAL mode updates (Firebase)
  refs.totals.child("total").transaction(cur => (cur||0) + pounds);
  refs.totals.child("supports").transaction(cur => (cur||0) + 1);
  if(type === "time") refs.totals.child("time").transaction(cur => (cur||0) + pounds);
  if(type === "comfort") refs.totals.child("comfort").transaction(cur => (cur||0) + 1);

  addMyCoins(influence);
  logAction(msg);

  // Instant UI update (so it feels responsive even before Firebase syncs)
  if(!localMode){
    // if totals listener hasn't fired yet, this prevents "nothing happened" feeling
    if(!live.totals) live.totals = { total:0, supports:0, time:0, comfort:0 };
    live.totals.total += pounds;
    live.totals.supports += 1;
    if(type === "time") live.totals.time += pounds;
    if(type === "comfort") live.totals.comfort += 1;
    renderTotals(live.totals);
  }
}

}

// Create a new decision (question + options) from the dashboard inputs
function createDecision(){
  if(!isHost()) return alert("Host only.");

  const q = (document.getElementById("cdQuestion")?.value || "").trim();
  const raw = (document.getElementById("cdOptions")?.value || "").trim();

  if(q.length < 5) return alert("Write a clearer question (min 5 chars).");
  if(raw.length < 3) return alert("Add at least 2 options (one per line).");

  // One option per line, cleaned
  const lines = raw
    .split("\n")
    .map(s => s.trim())
    .filter(Boolean)
    .slice(0, 6); // keep it tight for v1

  if(lines.length < 2) return alert("Add at least 2 options.");

  // Build options object (a,b,c,d,e,f)
  const keys = ["a","b","c","d","e","f"];
  const options = {};
  lines.forEach((name, i) => {
    options[keys[i]] = { name, votes: 0 };
  });

  const newDecision = {
    endsAt: Date.now() + 15*60*1000, // 15 min
    closed: false,
    question: q,
    desc: "Spend Influence to vote. First 100 Influence is strongest; after that votes diminish.",
    options,
    usedBy: {}
  };

  // Clear inputs (nice UX)
  document.getElementById("cdQuestion").value = "";
  document.getElementById("cdOptions").value = "";

  // Save
  if(localMode){
    local.decision = newDecision;
    renderDecision(local.decision);
    tickCountdown();
    logAction(`🎛 Host created a new decision: "${q}"`);
    return;
  }

  refs.decision.set(newDecision);
  logAction(`🎛 Host created a new decision: "${q}"`);
}

// Extend current decision by N minutes
function extendDecision(minutes){
  if(!isHost()) return alert("Host only.");
  minutes = Number(minutes) || 0;
  if(minutes <= 0) return;

  if(localMode){
    local.decision.endsAt = (local.decision.endsAt || Date.now()) + minutes*60*1000;
    local.decision.closed = false;
    renderDecision(local.decision);
    tickCountdown();
    logAction(`⏳ Host extended decision by +${minutes} minutes`);
    return;
  }

  refs.decision.child("endsAt").transaction(cur => (cur || Date.now()) + minutes*60*1000);
  refs.decision.child("closed").set(false);
  logAction(`⏳ Host extended decision by +${minutes} minutes`);
}
  
function resetEvent(){
  if(!isHost()) return alert("Host only.");
  const msg = "🔁 Host reset the event";

  if(localMode){
    local.totals = { total:0, supports:0, time:0, comfort:0 };
    local.feed = [];

    local.decision.endsAt = Date.now() + 15*60*1000;

    local.bounties = [];

    logAction(msg);
    return;
  }

  refs.totals.set({ total:0, supports:0, time:0, comfort:0 });
  refs.feed.remove();
  refs.bounties.remove();
  refs.decision.update({
    closed: false,
    endsAt: Date.now() + 15*60*1000,
    usedBy: null
  });

  // reset votes in options
  refs.decision.child("options").get().then(snap=>{
    const obj = snap.val() || {};
    Object.keys(obj).forEach(k=>{
      refs.decision.child("options").child(k).child("votes").set(0);
    });
  });

  logAction(msg);
}

// Host: approve the latest pending idea
function approveLatestBounty(){
  if(!isHost()) return alert("Host only.");
  if(localMode){
    const idx = (local.bounties || []).findIndex(x => (x.status || "pending") !== "approved");
    if(idx === -1) return alert("No pending ideas.");
    local.bounties[idx].status = "approved";
    renderBounties(local.bounties);
    logAction("✅ Host approved an idea");
    return;
  }

  // Firebase: find latest pending and mark approved
  refs.bounties
    .orderByChild("ts")
    .limitToLast(20)
    .once("value")
    .then(snap=>{
      const obj = snap.val() || {};
      const list = Object.entries(obj)
        .map(([id,v]) => ({ id, ...v }))
        .sort((a,b)=> (b.ts||0)-(a.ts||0));
      const pending = list.find(x => (x.status || "pending") !== "approved");
      if(!pending) return alert("No pending ideas.");
      refs.bounties.child(pending.id).child("status").set("approved");
      logAction("✅ Host approved an idea");
    });
}

function announceWinner(dec){
  const options = dec.options || {};
  let bestKey = null;
  let bestVotes = -1;
  Object.entries(options).forEach(([k,o])=>{
    const v = o.votes || 0;
    if(v > bestVotes){ bestVotes = v; bestKey = k; }
  });
  if(!bestKey || bestVotes <= 0){
    setAlert("🏁 Decision closed — no votes yet");
    return;
  }
  const winner = options[bestKey].name || bestKey;
  setAlert(`🏁 Decision closed — winner: ${winner}`);
}

// Vote logic (Influence spend with cap + diminishing returns)
function voteOption(optionKey, spend){
  const name = getUserName();
  if(name === "Anonymous") return alert("Set a display name first.");

  const dec = getDecision();
  if(dec.closed) return alert("Decision is closed.");

  spend = Number(spend) || 0;
  if(spend <= 0) return;

  const coins = getMyCoins();
  if(coins < spend) return alert("Not enough Influence.");

  const used = getMyUsedThisDecision();
  if(used + spend > CAP_PER_DECISION){
    return alert(`Cap reached. You can spend ${Math.max(0, CAP_PER_DECISION - used)} more this decision.`);
  }

  const addedVotes = votePower(spend);

  // Spend influence locally/UI
  setMyCoins(coins - spend);

  // LOCAL MODE
  if(localMode){
    if(!local.decision.options[optionKey]) return alert("Option not found.");
    local.decision.options[optionKey].votes = (local.decision.options[optionKey].votes||0) + addedVotes;
    if(!local.decision.usedBy) local.decision.usedBy = {};
    local.decision.usedBy[name] = (local.decision.usedBy[name]||0) + spend;

    renderDecision(local.decision);
    renderCoins();
    logAction(`🗳️ ${name} voted (+${addedVotes})`);
    return;
  }

  // GLOBAL MODE (Firebase)
  refs.decision.child("options").child(optionKey).child("votes")
    .transaction(cur => (cur||0) + addedVotes);

  refs.decision.child("usedBy").child(name)
    .transaction(cur => (cur||0) + spend);

  logAction(`🗳️ ${name} voted (+${addedVotes})`);
}

const DIR_KEY = "cv_directory_pass1";
let selectedCreator = null;

function normHandle(h){
  h = String(h||"").trim().toLowerCase();
  h = h.replace(/^@+/, "");
  h = h.replace(/[^a-z0-9_]/g, "").slice(0, 20);
  return h;
}

function getDirectoryLocal(){
  try{ return JSON.parse(localStorage.getItem(DIR_KEY) || "[]"); }
  catch(e){ return []; }
}
function setDirectoryLocal(list){
  localStorage.setItem(DIR_KEY, JSON.stringify(list || []));
}

function chaosScoreFromTotals(t){
  const total = Number(t?.total||0);
  const supports = Number(t?.supports||0);
  const time = Number(t?.time||0);
  const comfort = Number(t?.comfort||0);
  // v1 chaos: tune later
  return Math.round(total*3 + supports*2 + time*1 + comfort*6);
}

function sessionKey(handle){ return `cv_session_${handle}`; }

function startSession(){
  if(!isHost()) return alert("Host only.");
  const handle = normHandle(document.getElementById("cpHandleInput")?.value || getUserName());
  if(!handle || handle.length < 2) return alert("Add a valid @handle first.");
  localStorage.setItem(sessionKey(handle), JSON.stringify({ startedAt: Date.now(), endedAt: null }));
  setAlert("🟢 Session started");
}

function endSession(){
  if(!isHost()) return alert("Host only.");
  const handle = normHandle(document.getElementById("cpHandleInput")?.value || getUserName());
  if(!handle || handle.length < 2) return alert("Add a valid @handle first.");
  const cur = JSON.parse(localStorage.getItem(sessionKey(handle)) || "null") || { startedAt: Date.now() };
  cur.endedAt = Date.now();
  localStorage.setItem(sessionKey(handle), JSON.stringify(cur));
  setAlert("🔴 Session ended");
}

function minutesElapsed(handle){
  try{
    const s = JSON.parse(localStorage.getItem(sessionKey(handle)) || "null");
    if(!s || !s.startedAt) return 0;
    const end = s.endedAt || Date.now();
    return Math.max(0, Math.round((end - s.startedAt)/60000));
  }catch(e){ return 0; }
}

async function publishCreatorPage(){
  if(!isHost()) return alert("Host only.");
  const d = getCreatorDraft();
  if(!d.name || d.name.length < 2) return alert("Add a creator name.");
  if(!d.handle || d.handle.length < 2) return alert("Add a valid @handle (letters/numbers/_).");

  const totals = getCurrentTotals();
  const chaos = chaosScoreFromTotals(totals);
  const mins = minutesElapsed(d.handle);
  const ppm = mins > 0 ? (Number(totals.total||0) / mins) : 0;

  const creator = {
    ...d,
    stats: {
      total: Number(totals.total||0),
      supports: Number(totals.supports||0),
      time: Number(totals.time||0),
      comfort: Number(totals.comfort||0),
      chaos: Number(chaos||0),
      minutes: Number(mins||0),
      ppm: Math.round(ppm*100)/100
    }
  };

  // Always publish to local directory (so GitHub Pages works)
  const list = getDirectoryLocal();
  const idx = list.findIndex(x => x.handle === creator.handle);
  if(idx >= 0) list[idx] = { ...list[idx], ...creator };
  else list.unshift(creator);
  setDirectoryLocal(list.slice(0, 300));

  // Optional: also publish to Firebase directory if Firebase is configured & connected
  try{
    if(!localMode && typeof firebase !== "undefined" && firebase?.database){
      const db = firebase.database();
      db.ref("cv_directory/creators").child(creator.handle).set(creator);
    }
  }catch(e){ /* ignore */ }

  setAlert("📣 Published");
  refreshDirectory();
}

async function refreshDirectory(){
  const search = document.getElementById("discoverSearch");
  if(search && !search.__wired){
    search.__wired = true;
    search.addEventListener("input", ()=> refreshDirectory());
  }

  // Local directory always exists
  let list = getDirectoryLocal();

  // Optional: merge Firebase directory entries (if available)
  try{
    if(!localMode && typeof firebase !== "undefined" && firebase?.database){
      const db = firebase.database();
      const snap = await db.ref("cv_directory/creators").limitToLast(300).get();
      const obj = snap.val() || {};
      const fb = Object.values(obj);
      // Merge by handle (Firebase overrides local if newer)
      const byHandle = {};
      list.forEach(c=> byHandle[c.handle] = c);
      fb.forEach(c=>{
        if(!c || !c.handle) return;
        const cur = byHandle[c.handle];
        if(!cur || (c.updatedAt||0) >= (cur.updatedAt||0)) byHandle[c.handle] = c;
      });
      list = Object.values(byHandle).sort((a,b)=> (b.updatedAt||0)-(a.updatedAt||0));
    }
  }catch(e){ /* ignore */ }

  renderCreatorFeed(list);
  renderLeaderboards(list);
}
